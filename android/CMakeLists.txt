# Grand Chase 1 (recomp) para Android: libSDL3.so + libmain.so (runtime + codigo recompilado + plataforma SDL).
# Uso: android/build_apk.sh  (chama o cmake do NDK para cada ABI)
cmake_minimum_required(VERSION 3.22)
project(gc1 C)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/SDL3-3.4.20-src" SDL3-build EXCLUDE_FROM_ALL)
file(GLOB BLOCKS "${ROOT}/gen/blocks_*.c")
add_library(main SHARED ${BLOCKS} "${ROOT}/gen/tables.c" "${ROOT}/runtime/cpu.c" "${ROOT}/runtime/hle.c" "${ROOT}/runtime/audio_policy.c" "${ROOT}/platform/main_sdl.c")
target_include_directories(main PRIVATE "${ROOT}/runtime" "${ROOT}/platform")
target_compile_options(main PRIVATE -O1 -w -DGC1_CHECK_MEM)
target_link_libraries(main PRIVATE SDL3::SDL3 log android)
